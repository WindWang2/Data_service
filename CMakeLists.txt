cmake_minimum_required(VERSION 3.24)
project(PaleoDataService VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Sql Widgets)
find_package(ZLIB REQUIRED)
include(cmake/SeismicEngine.cmake)
file(GLOB PARSER_SOURCES CONFIGURE_DEPENDS src/parsers/*.cpp)
file(GLOB STORE_SOURCES CONFIGURE_DEPENDS src/store/*.cpp)
file(GLOB FLOW_SOURCES CONFIGURE_DEPENDS src/flow/*.cpp)
file(GLOB SERVICE_SOURCES CONFIGURE_DEPENDS src/services/*.cpp src/adapters/*.cpp)
add_library(paleo_data_service STATIC ${PARSER_SOURCES} ${STORE_SOURCES} ${FLOW_SOURCES} ${SERVICE_SOURCES} vendor/paleo_io/ziparchive.cpp)
target_include_directories(paleo_data_service PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE src vendor/paleo_io vendor/sbm/src vendor/glm vendor/segyio/include)
target_compile_definitions(paleo_data_service PRIVATE PALEO_HAVE_ZLIB)
target_link_libraries(paleo_data_service PRIVATE Qt6::Core Qt6::Gui Qt6::Sql ZLIB::ZLIB pds_sbm)
if(WIN32)
  target_link_libraries(paleo_data_service PRIVATE bcrypt)
endif()
add_library(Paleo::paleo_data_service ALIAS paleo_data_service)
if(MSVC)
  target_compile_options(paleo_data_service PRIVATE /utf-8 /W4)
  add_compile_options(/utf-8)
endif()
file(GLOB DEMO_SOURCES CONFIGURE_DEPENDS demo/*.cpp demo/*.h)
if(DEMO_SOURCES)
  add_executable(paleo_data_demo WIN32 ${DEMO_SOURCES})
  target_link_libraries(paleo_data_demo PRIVATE paleo_data_service Qt6::Widgets)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/service_cli.cpp")
  add_executable(paleo_data_cli tools/service_cli.cpp)
  target_link_libraries(paleo_data_cli PRIVATE paleo_data_service Qt6::Core)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/seismic_acceptance.cpp")
  add_executable(seismic_acceptance tools/seismic_acceptance.cpp)
  target_link_libraries(seismic_acceptance PRIVATE paleo_data_service Qt6::Core)
endif()
add_executable(segyio_acceptance tools/segyio_acceptance.cpp)
target_link_libraries(segyio_acceptance PRIVATE paleo_data_service Qt6::Core)
add_executable(timeslice_benchmark tools/timeslice_benchmark.cpp)
target_link_libraries(timeslice_benchmark PRIVATE paleo_data_service Qt6::Core)
include(CTest)
if(BUILD_TESTING)
  add_executable(segyio_bounds_tests tests/segyio_bounds_tests.cpp)
  target_include_directories(segyio_bounds_tests PRIVATE vendor/segyio/include)
  target_link_libraries(segyio_bounds_tests PRIVATE pds_segyio)
  add_test(NAME segyio_bounds_tests COMMAND segyio_bounds_tests)
endif()
foreach(test_name IN ITEMS aux_parser_tests text_parser_tests seismic_service_tests seismic_hd_tests timeslice_batch_tests spreadsheet_xml_tests core_asset_tests format_compatibility_tests segyio_access_tests seismic_processing_tests import_performance_tests file_digest_tests)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${test_name}.cpp")
    add_executable(${test_name} tests/${test_name}.cpp)
    target_include_directories(${test_name} PRIVATE src)
    target_link_libraries(${test_name} PRIVATE paleo_data_service Qt6::Core Qt6::Sql)
    add_test(NAME ${test_name} COMMAND ${test_name})
  endif()
endforeach()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/service_tests.cpp")
  add_executable(service_tests tests/service_tests.cpp)
  target_link_libraries(service_tests PRIVATE paleo_data_service Qt6::Core)
  add_test(NAME service_contract COMMAND service_tests)
endif()
install(TARGETS paleo_data_service pds_sbm pds_segyio EXPORT PaleoDataServiceTargets ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(EXPORT PaleoDataServiceTargets FILE PaleoDataServiceTargets.cmake NAMESPACE Paleo:: DESTINATION lib/cmake/PaleoDataService)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/PaleoDataServiceConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/PaleoDataServiceConfig.cmake" INSTALL_DESTINATION lib/cmake/PaleoDataService)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/PaleoDataServiceConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/PaleoDataServiceConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/PaleoDataServiceConfigVersion.cmake" DESTINATION lib/cmake/PaleoDataService)
add_executable(facies_feature_example examples/facies_feature/main.cpp)
target_link_libraries(facies_feature_example PRIVATE Paleo::paleo_data_service Qt6::Core)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tools/create_api_demo_project.cpp")
  add_executable(create_api_demo_project tools/create_api_demo_project.cpp)
  target_link_libraries(create_api_demo_project PRIVATE Paleo::paleo_data_service Qt6::Core)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/api_demo_tests.cpp")
  add_executable(api_demo_tests tests/api_demo_tests.cpp demo/apidemo.cpp demo/apidemo.h demo/plotwidgets.cpp demo/plotwidgets.h)
  target_include_directories(api_demo_tests PRIVATE demo)
  target_link_libraries(api_demo_tests PRIVATE Paleo::paleo_data_service Qt6::Widgets)
  add_test(NAME api_demo_contract COMMAND api_demo_tests)
endif()
if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/image_preview_tests.cpp")
  add_executable(image_preview_tests tests/image_preview_tests.cpp demo/plotwidgets.cpp)
  target_include_directories(image_preview_tests PRIVATE demo)
  target_link_libraries(image_preview_tests PRIVATE Paleo::paleo_data_service Qt6::Widgets)
  add_test(NAME image_preview_contract COMMAND image_preview_tests)

if(BUILD_TESTING AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/seismic_ui_tests.cpp")
  add_executable(seismic_ui_tests tests/seismic_ui_tests.cpp demo/apidemo.cpp demo/apidemo.h demo/plotwidgets.cpp)
